# Yêu cầu môi trường cho NetworkProxySystem
#
# Đây là dự án Java/Maven, không phải dự án Python. Maven sẽ tự tải các
# thư viện được khai báo trong proxy-server/pom.xml và admin-api/pom.xml.

Java Development Kit (JDK) >= 17
Apache Maven >= 3.9

# Cơ sở dữ liệu
Microsoft SQL Server 2019/2022/2025 (Express, Developer hoặc Enterprise)
sqlcmd (SQL Server Command Line Tools)

# Thư viện Maven của proxy-server (tự tải từ pom.xml):
org.apache.sshd:sshd-core:2.12.1
org.apache.sshd:sshd-common:2.12.1
org.junit.jupiter:junit-jupiter:5.11.0 (test)

# Thư viện Maven của admin-api (tự tải từ pom.xml):
org.springframework.boot:spring-boot-starter-web:3.3.5
org.springframework.boot:spring-boot-starter-jdbc:3.3.5
org.springframework.boot:spring-boot-starter-validation:3.3.5
org.springframework.security:spring-security-crypto
com.microsoft.sqlserver:mssql-jdbc:13.6.0.jre11

# Tùy chọn cho Windows Integrated Security:
# - Tải mssql-jdbc_auth-13.6.0.x64.dll và đặt trong java.library.path
#   https://repo.maven.apache.org/maven2/com/microsoft/sqlserver/mssql-jdbc_auth/13.6.0.x64/

# Tùy chọn cho chức năng SSH:
# - Tệp ~/.ssh/known_hosts hợp lệ
# - Khóa riêng SSH hoặc mật khẩu được cung cấp lúc chạy
